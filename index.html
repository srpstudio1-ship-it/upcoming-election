import time import pandas as pd from selenium import webdriver from selenium.webdriver.common.by import By from selenium.webdriver.support.ui import Select, WebDriverWait from selenium.webdriver.support import expected_conditions as EC from webdriver_manager.chrome import ChromeDriverManager from selenium.webdriver.chrome.service import Service def scrape_nagar_palika_votes(district_name="JALORE"): # Setup Chrome Driver service = Service(ChromeDriverManager().install()) driver = webdriver.Chrome(service=service) # Official SEC Rajasthan Download / Results Portal URL url = "https://sec.rajasthan.gov.in/se_pdfdownload.aspx" driver.get(url) wait = WebDriverWait(driver, 10) vote_data = [] try: # Select District district_dropdown = Select(wait.until(EC.presence_of_element_located((By.ID, "ddlDistrict")))) district_dropdown.select_by_visible_text(district_name) time.sleep(2) # Select Municipality (Nagar Palika / Parishad) body_dropdown = Select(wait.until(EC.presence_of_element_located((By.ID, "ddlLocalBody")))) body_options = [opt.text for opt in body_dropdown.options if "Select" not in opt.text] for body in body_options: body_dropdown.select_by_visible_text(body) time.sleep(2) # Click Search/View Results search_btn = driver.find_element(By.ID, "btnSearch") search_btn.click() time.sleep(3) # Extract Results Table rows = driver.find_elements(By.XPATH, "//table[@id='gvResults']//tr[position()>1]") for row in rows: cols = row.find_elements(By.TAG_NAME, "td") if len(cols) >= 6: vote_data.append({ "District": district_name, "Nagar Palika": body, "Ward No": cols[0].text.strip(), "Candidate Name": cols[1].text.strip(), "Party": cols[2].text.strip(), "Votes Secured": cols[3].text.strip(), "Margin / Result": cols[4].text.strip() }) except Exception as e: print(f"Scraper error: {e}") finally: driver.quit() df = pd.DataFrame(vote_data) df.to_csv(f"{district_name}_Nagar_Palika_Votes.csv", index=False) print(f"Exported {len(vote_data)} records to {district_name}_Nagar_Palika_Votes.csv") if __name__ == "__main__": scrape_nagar_palika_votes("JALORE")
